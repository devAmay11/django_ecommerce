# 1. Session-Based Cart Management
# The session stores the cart data as a dictionary, where the keys are the product IDs and the values are the quantities of each product.

# views.py
from django.http import JsonResponse

def add_to_cart(request, product_id):
    product = Product.objects.get(id=product_id)
    cart = request.session.get('cart', {})

    if str(product_id) in cart:
        cart[str(product_id)] += 1
    else:
        cart[str(product_id)] = 1

    request.session['cart'] = cart
    cart_count = sum(cart.values())
    
    return JsonResponse({'cart_count': cart_count})

# Flow:
# The user clicks "Add to Cart."
# The add_to_cart view is triggered, retrieving the product by its ID.
# The session is checked for the existing cart. If it doesn't exist, an empty cart (a dictionary) is created.
# If the product is already in the cart, its quantity is incremented. Otherwise, it is added with a quantity of 1.
# The cart is saved back into the session (request.session['cart'] = cart).
# The user is redirected back to the product list (or another relevant page).



# 2. Displaying Cart Count on Every Page
# To display the total number of items in the cart on every page, you can use Django’s context processors. This makes the cart count available as a global variable that can be accessed in all templates.

# Custom Context Processor:

# context_processors.py
def cart_item_count(request):
    cart = request.session.get('cart', {})  # Retrieve the cart from the session
    cart_count = sum(cart.values())  # Calculate total quantity in the cart
    return {
        'cart_count': cart_count  # Return cart count as part of the context
    }

# Flow:

# Every time a page is loaded, the context processor cart_item_count runs.
# It retrieves the cart from the session (or an empty cart if none exists).
# The total count of items in the cart is calculated by summing the quantities of all products (sum(cart.values())).
# This cart_count is added to the context, making it available in all templates.

# In settings.py, you must add this custom context processor to ensure it runs on every request.
